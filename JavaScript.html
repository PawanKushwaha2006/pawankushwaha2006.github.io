01

Introduction to JavaScript

JavaScript was created by Brendan Eich in 1995 in just 10 days while he worked at Netscape. Originally called Mocha, then LiveScript, it became JavaScript as a marketing move to ride Java's popularity — despite the two having almost nothing in common.

Today, JavaScript is the only language that runs natively in every browser. With Node.js, it runs on servers too. It powers front-end UIs, back-end APIs, mobile apps (React Native), desktop apps (Electron), and even machine learning (TensorFlow.js).

Variables

Variables store data. JavaScript has three ways to declare them: var (old), let (modern, reassignable), and const (modern, constant reference).

// ── var (old, avoid in modern code) ──
var name = "Alice";   // function-scoped, hoisted
var name = "Bob";    // re-declaration allowed (dangerous!)

// ── let (modern, prefer for mutable values) ──
let age = 25;
age = 26;             // ✓ reassignment OK
// let age = 30;      ✗ re-declaration not allowed

// ── const (modern, prefer for everything else) ──
const PI = 3.14159;
// PI = 3;            ✗ cannot reassign

const user = { name: "Alice" };
user.name = "Bob";    // ✓ object properties can still change
// user = {};         ✗ but binding itself cannot be reassigned

// ── Naming Rules ──
let camelCase   = "standard for variables";
const UPPER_CASE = "convention for constants";
let _private    = "underscore prefix convention";
// let 1invalid   = ✗ cannot start with a digit

Data Types

JavaScript has 8 data types: 7 primitives (immutable values) and 1 object type (mutable reference).

// ── Primitives ──
let num   = 42;                    // Number (int + float)

// ── Object ──
let obj  = { key: "value" };      // Object
let arr  = [1, 2, 3];              // Array (type of Object)

Operators

// ── Arithmetic ──
10 + 3    // 13  (addition)
10 - 3    // 7   (subtraction)
10 * 3    // 30  (multiplication)
10 / 3    // 3.333...  (division)
10 % 3    // 1   (modulus / remainder)
2 ** 8    // 256 (exponentiation)

// ── Assignment ──
let x = 10;
x += 5;   // x = 15
x -= 3;   // x = 12
x *= 2;   // x = 24
x /= 4;   // x = 6
x++;       // x = 7  (post-increment)
x--;       // x = 6  (post-decrement)

// ── Comparison ──
5 == "5"   // true  (loose equality — type coercion)
5 === "5"  // false (strict equality — no coercion)
5 != "5"   // false
5 !== "5"  // true  ← always use !== over !=
10 > 5    // true
10 >= 10  // true

// ── Logical ──
true && false    // false (AND)
true || false    // true  (OR)
!true            // false (NOT)

// ── Nullish Coalescing (??) — ES2020 ──
let val = null ?? "default";   // "default"
let val2 = 0 ?? "default";    // 0  (0 is not null/undefined)

// ── Optional Chaining (?.) — ES2020 ──
let user = { profile: { name: "Alice" } };
user?.profile?.name        // "Alice"
user?.address?.city        // undefined (no error)

// ── Ternary Operator ──
let age = 20;
let status = age >= 18 ? "adult" : "minor"; // "adult"
Always Use === Not == == performs type coercion, leading to unexpected results like 0 == false // true or "" == false // true. Always use === for comparisons.

Loops

// ── for loop ──
for (let i = 0; i < 5; i++) {
  console.log(i);    // 0 1 2 3 4
}

// ── while loop ──
let count = 0;
while (count < 3) {
  console.log(count++);
}

// ── do...while (always runs at least once) ──
let n = 10;
do {
  console.log(n);
  n++;
} while (n < 3);    // runs once (n=10, condition false)

// ── for...of (arrays, strings, iterables) ──
const fruits = ["apple", "banana", "cherry"];
for (const fruit of fruits) {
  console.log(fruit);
}

// ── for...in (object keys) ──
const user = { name: "Alice", age: 25, city: "Delhi" };
for (const key in user) {
  console.log(`${key}: ${user[key]}`);
}

// ── break and continue ──
for (let i = 0; i < 10; i++) {
  if (i === 3) continue;   // skip 3
  if (i === 6) break;      // stop at 6
  console.log(i);           // 0 1 2 4 5
}
07

Functions

Functions are reusable blocks of code. JavaScript has several ways to define them, each with subtle differences in behavior.

// ── Function Declaration (hoisted) ──
function greet(name) {
  return `Hello, ${name}!`;
}
greet("Alice");   // "Hello, Alice!"

// ── Function Expression (not hoisted) ──
const greet = function(name) {
  return `Hello, ${name}!`;
};

// ── Arrow Function (ES6) — concise syntax ──
const greet = (name) => `Hello, ${name}!`;
const double = n => n * 2;         // single param, no parens needed
const getObj = () => ({ key: 1 });  // wrap object in parens

// ── Default Parameters ──
function greet(name = "stranger") {
  return `Hello, ${name}!`;
}
greet();         // "Hello, stranger!"
greet("Bob");    // "Hello, Bob!"

// ── Rest Parameters ──
function sum(...nums) {
  return nums.reduce(total, n) => total + n, 0);
}
sum(1, 2, 3, 4);   // 10

// ── Immediately Invoked Function Expression (IIFE) ──
(function() {
  console.log("Runs immediately!");
})();

// ── Higher Order Functions ──
function applyTwice(fn, value) {
  return fn(fn(value));
}
applyTwice(x => x * 2, 3);  // 12

// ── Closure ──
function counter() {
  let count = 0;
  return () => ++count;   // inner fn "closes over" count
}
const increment = counter();
increment();  // 1
increment();  // 2
Arrow Functions and this Arrow functions do NOT have their own this. They inherit this from the surrounding lexical scope. Use regular functions for object methods; use arrow functions for callbacks.

Scope & Hoisting

// ── Global Scope ──
let globalVar = "I'm everywhere";

function test() {
  // ── Function Scope ──
  var funcScoped = "only in this function";

  if (true) {
    // ── Block Scope ──
    let blockScoped = "only in this block";
    const alsoBlock = "same";
    var  leaksOut   = "var leaks to function!";
  }

  console.log(leaksOut);    // ✓ works (var leaks)
  // console.log(blockScoped); ✗ ReferenceError
}

// ── Hoisting ──
// var is hoisted and initialized as undefined
console.log(hoisted);        // undefined (no error!)
var hoisted = "value";

// let/const are hoisted but NOT initialized (Temporal Dead Zone)
// console.log(notHoisted);  ✗ ReferenceError
let notHoisted = "value";

// Function declarations are fully hoisted
sayHi();    // ✓ works before declaration
function sayHi() { console.log("Hi!"); }
09

Arrays

Arrays store ordered lists of values. JavaScript arrays are dynamic — they can hold mixed types and grow/shrink freely.

const nums = [10, 20, 30, 40, 50];

// ── Access & Modify ──
nums[0]           // 10  (first element)
nums.length        // 5
nums[0] = 99;     // mutate first element

// ── Add / Remove ──
nums.push(60);     // add to end → [10,20,30,40,50,60]
nums.pop();        // remove from end → returns 60
nums.unshift(0);   // add to start → [0,10,20,30,40,50]
nums.shift();      // remove from start → returns 0
nums.splice(1, 2); // remove 2 from index 1

// ── Search ──
nums.indexOf(30)          // 2
nums.includes(20)         // true
nums.find(n => n > 25)    // 30 (first match)
nums.findIndex(n => n > 25) // 2

// ── Iteration ──
nums.forEach(n => console.log(n));

// ── Transformation (returns new array) ──
nums.map(n => n * 2)       // [20,40,60,80,100]
nums.filter(n => n > 20)   // [30,40,50]
nums.reduce((acc, n) => acc + n, 0) // 150 (sum)
nums.slice(1, 3)            // [20,30] (non-mutating)
nums.sort((a, b) => a - b) // ascending sort
nums.reverse()              // reverse in place
nums.flat()                 // flatten nested arrays
nums.every(n => n > 0)     // true if ALL pass
nums.some(n => n > 40)     // true if ANY pass
[...nums, 60, 70]           // spread to combine arrays
[...new Set(nums)]          // deduplicate array

Objects

Objects store key-value pairs. They're the backbone of JavaScript — almost everything in JS is an object.

// ── Object Literal ──
const user = {
  name:   "Alice",
  age:    25,
  isAdmin: false,
  address: {
    city:    "Delhi",
    country: "India"
  },
  greet() {
    return `Hi, I'm ${this.name}`;
  }
};

// ── Access ──
user.name            // "Alice"  (dot notation)
user["age"]          // 25  (bracket notation)
user.address.city    // "Delhi"
user.greet()         // "Hi, I'm Alice"

// ── Add / Update / Delete ──
user.email = "alice@example.com";  // add
user.age   = 26;                    // update
delete user.isAdmin;               // delete

// ── Checking Keys ──
"name" in user            // true
user.hasOwnProperty("age") // true

// ── Object Methods ──
Object.keys(user)          // ["name","age","address","greet"]
Object.values(user)        // ["Alice", 25, {...}, fn]
Object.entries(user)       // [["name","Alice"], ["age",25], ...]
Object.assign({}, user)    // shallow copy
Object.freeze(user)        // make object immutable

// ── Spread (shallow copy / merge) ──
const updated = { ...user, age: 30, role: "admin" };

// ── Computed Property Names ──
const key = "name";
const obj = { [key]: "Alice" };  // { name: "Alice" }

// ── Shorthand Properties ──
const name = "Bob", age = 30;
const person = { name, age };  // { name: "Bob", age: 30 }

Destructuring

Destructuring lets you unpack values from arrays or properties from objects into variables with a clean, readable syntax.

// ── Array Destructuring ──
const [a, b, c] = [10, 20, 30];
// a=10, b=20, c=30

const [first, , third] = [1, 2, 3];  // skip middle

const [x = 5, y = 10] = [100];
// x=100, y=10 (default used)

// Swap variables
let p = 1, q = 2;
[p, q] = [q, p];  // p=2, q=1

// ── Object Destructuring ──
const user = { name: "Alice", age: 25, city: "Delhi" };

const { name, age } = user;
// name="Alice", age=25

// Rename during destructuring
const { name: userName, age: userAge } = user;

// Default values
const { name, role = "guest" } = user;

// Nested destructuring
const { address: { city, country } } = {
  address: { city: "Mumbai", country: "India" }
};

// ── In Function Parameters ──
function displayUser({ name, age = 0 }) {
  console.log(`${name} is ${age}`);
}
displayUser({ name: "Bob", age: 30 });

Spread & Rest

// ── Spread (...) — expand iterable into individual items ──

// Copy array
const original = [1, 2, 3];
const copy     = [...original];

// Merge arrays
const merged = [...[1, 2], ...[3, 4]];  // [1,2,3,4]

// Spread in function call
Math.max(...[5, 1, 9, 3]);   // 9

// Copy and merge objects
const defaults = { theme: "dark", lang: "en" };
const settings = { ...defaults, lang: "hi" };
// { theme: "dark", lang: "hi" }

// ── Rest (...) — collect remaining items ──

// Rest in function parameters
function sum(first, ...rest) {
  console.log(first);  // 1
  console.log(rest);   // [2,3,4]
  return first + rest.reduce((a, b) => a + b, 0);
}
sum(1, 2, 3, 4);  // 10

// Rest in destructuring
const [head, ...tail] = [10, 20, 30, 40];
// head=10, tail=[20,30,40]

const { name, ...rest } = { name: "Alice", age: 25, city: "Delhi" };
// name="Alice", rest={ age:25, city:"Delhi" }

Classes & OOP

JavaScript classes (ES6+) provide syntactic sugar over prototype-based inheritance. They make OOP patterns cleaner and more familiar.

// ── Class Definition ──
class Animal {
  // Private field (ES2022)
  #sound;

  constructor(name, sound) {
    this.name  = name;
    this.#sound = sound;
  }

  // Instance method
  speak() {
    return `${this.name} says ${this.#sound}!`;
  }

  // Static method
  static kingdom() { return "Animalia"; }

  // Getter
  get info() { return `I am ${this.name}`; }

  // Setter
  set nickname(val) { this.nick = val; }
}

// ── Inheritance ──
class Dog extends Animal {
  constructor(name, breed) {
    super(name, "Woof");  // call parent constructor
    this.breed = breed;
  }

  // Override parent method
  speak() {
    return `${super.speak()} (${this.breed})`;
  }

  fetch() { return `${this.name} fetches the ball!`; }
}

// ── Usage ──
const dog = new Dog("Rex", "Labrador");
dog.speak();                // "Rex says Woof! (Labrador)"
dog.info;                   // "I am Rex"
Animal.kingdom();           // "Animalia"
dog instanceof Dog;        // true
dog instanceof Animal;     // true

Promises & Async/Await

Asynchronous code lets JavaScript do things like fetch data from a server without freezing the browser. Promises represent a future value; async/await is the modern way to handle them.

// ── Callback (old way — "callback hell") ──
setTimeout(() => {
  console.log("After 1 second");
}, 1000);

// ── Promise ──
const promise = new Promise((resolve, reject) => {
  const success = true;
  if (success) resolve("Done!");
  else         reject("Failed!");
});

promise
  .then(result => console.log(result))  // "Done!"
  .catch(err  => console.error(err))
  .finally(() => console.log("Always runs"));

// ── Async / Await (modern, preferred) ──
async function getUser(id) {
  try {
    const response = await fetch(`/api/users/${id}`);
    if (!response.ok) throw new Error("Not found");
    const data = await response.json();
    return data;
  } catch (err) {
    console.error(err.message);
  }
}

// ── Promise Combinators ──

// All resolve (fail if any rejects)
await Promise.all([fetchA(), fetchB(), fetchC()]);

// First one to settle (resolve or reject)
await Promise.race([fast(), slow()]);

// All settle, get all results (with status)
await Promise.allSettled([a(), b()]);

// First one to resolve successfully
await Promise.any([a(), b()]);

// ── Parallel vs Sequential ──
// Sequential (slower — waits for each)
const user  = await getUser(1);
const posts = await getPosts(1);

// Parallel (faster — run at the same time)
const [user, posts] = await Promise.all([getUser(1), getPosts(1)]);

DOM Manipulation

The DOM (Document Object Model) is a tree-like representation of your HTML page. JavaScript can read and change it to make pages dynamic.

// ── Selecting Elements ──
document.getElementById("myId")          // single element by ID
document.querySelector(".myClass")       // first match (CSS selector)
document.querySelectorAll("p")           // NodeList of all <p> tags
document.getElementsByClassName("btn")  // HTMLCollection

// ── Reading & Changing Content ──
const el = document.querySelector("#title");

el.textContent     // get text (safe, no HTML)
el.textContent = "New Title";  // set text
el.innerHTML   = "<b>Bold</b>"; // set HTML (XSS risk!)
el.innerText   // like textContent but CSS-aware

// ── Attributes ──
el.getAttribute("href")
el.setAttribute("href", "https://example.com")
el.removeAttribute("disabled")
el.id    // shortcut for id attribute
el.href  // shortcut for href

// ── CSS Styles ──
el.style.color           = "red";
el.style.backgroundColor = "blue";
el.style.display         = "none";

// ── CSS Classes ──
el.classList.add("active");
el.classList.remove("hidden");
el.classList.toggle("open");
el.classList.contains("active");  // true/false

// ── Creating & Inserting Elements ──
const div = document.createElement("div");
div.textContent = "Hello!";
div.classList.add("box");
document.body.appendChild(div);       // add to end
document.body.prepend(div);          // add to start
el.insertAdjacentHTML("afterend", "<p>Hi</p>");

// ── Removing Elements ──
el.remove();                 // remove element itself
el.parentNode.removeChild(el);  // old way

// ── Traversal ──
el.parentElement
el.children          // child elements
el.firstElementChild
el.lastElementChild
el.nextElementSibling
el.previousElementSibling

Events

// ── addEventListener (recommended) ──
const btn = document.querySelector("#myBtn");

btn.addEventListener("click", (event) => {
  console.log("Clicked!");
  console.log(event.target);       // element that was clicked
  event.preventDefault();          // stop default (e.g. form submit)
  event.stopPropagation();         // stop event bubbling up
});

// ── Common Event Types ──
// Mouse:    click, dblclick, mousedown, mouseup, mousemove, mouseover
// Keyboard: keydown, keyup, keypress
// Form:     submit, change, input, focus, blur
// Window:   load, resize, scroll, DOMContentLoaded

// ── Keyboard Events ──
document.addEventListener("keydown", (e) => {
  if (e.key === "Enter")  console.log("Enter pressed");
  if (e.key === "Escape") console.log("Escape pressed");
  if (e.ctrlKey && e.key === "s") console.log("Ctrl+S");
});

// ── Form Events ──
const form  = document.querySelector("form");
const input = document.querySelector("input");

form.addEventListener("submit", (e) => {
  e.preventDefault();
  console.log("Form submitted");
});

input.addEventListener("input", (e) => {
  console.log(e.target.value);   // live input value
});

// ── Event Delegation (efficient for many children) ──
document.querySelector("ul").addEventListener("click", (e) => {
  if (e.target.tagName === "LI") {
    console.log("List item clicked:", e.target.textContent);
  }
});

// ── Once option (auto-remove after one trigger) ──
btn.addEventListener("click", handleClick, { once: true });

// ── Remove event listener ──
btn.removeEventListener("click", handleClick);

Modules (ES6)

Modules split code into separate files with explicit imports and exports. This keeps code organized and avoids naming collisions.

// ── math.js (named exports) ──
export const PI = 3.14159;
export function add(a, b) { return a + b; }
export function multiply(a, b) { return a * b; }

// ── user.js (default export) ──
export default class User {
  constructor(name) { this.name = name; }
}

// ── main.js (importing) ──

// Named imports
import { PI, add } from "./math.js";

// Rename on import
import { add as sum } from "./math.js";

// Import everything
import * as Math from "./math.js";
Math.add(2, 3);  // 5

// Default import (any name)
import User from "./user.js";
import Person from "./user.js";  // also valid

// Mix default and named
import User, { createUser } from "./user.js";

// Dynamic import (lazy loading)
const module = await import("./math.js");

<!-- In HTML, use type="module" -->
<script type="module" src="main.js"></script>

Error Handling

// ── try / catch / finally ──
try {
  const data = JSON.parse("invalid json {");
} catch (err) {
  console.error(err.name);     // "SyntaxError"
  console.error(err.message);  // description
  console.error(err.stack);    // stack trace
} finally {
  console.log("Always runs");  // cleanup code here
}

// ── Throwing Errors ──
function divide(a, b) {
  if (b === 0) throw new Error("Division by zero");
  return a / b;
}

// ── Built-in Error Types ──
new Error("Generic error")
new TypeError("Wrong type")
new RangeError("Out of range")
new ReferenceError("Undefined variable")
new SyntaxError("Bad syntax")

// ── Custom Error Class ──
class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name  = "ValidationError";
    this.field = field;
  }
}

try {
  throw new ValidationError("Required", "email");
} catch (err) {
  if (err instanceof ValidationError) {
    console.log(`${err.field}: ${err.message}`);  // "email: Required"
  }
}

Web Storage

localStorage persists data even after the browser closes. sessionStorage clears when the tab closes. Both store string key-value pairs.

// ── localStorage ──
localStorage.setItem("theme", "dark");
localStorage.getItem("theme");           // "dark"
localStorage.removeItem("theme");
localStorage.clear();                    // remove ALL keys
localStorage.length;                     // number of items

// ── Store objects (must JSON stringify) ──
const user = { name: "Alice", age: 25 };
localStorage.setItem("user", JSON.stringify(user));
const loaded = JSON.parse(localStorage.getItem("user"));

// ── sessionStorage (same API, tab-scoped) ──
sessionStorage.setItem("token", "abc123");
sessionStorage.getItem("token");

// ── Cookies (for server communication) ──
document.cookie = "username=Alice; expires=Fri, 31 Dec 2025 23:59:59 GMT; path=/";
document.cookie;   // read all cookies (as a single string)

Fetch API

The Fetch API is the modern way to make HTTP requests from JavaScript. It returns Promises and replaces the older XMLHttpRequest.

// ── GET Request ──
async function getUsers() {
  const res   = await fetch("https://api.example.com/users");
  if (!res.ok) throw new Error(`HTTP Error: ${res.status}`);
  const data  = await res.json();
  return data;
}

// ── POST Request (send JSON) ──
async function createUser(user) {
  const res = await fetch("https://api.example.com/users", {
    method:  "POST",
    headers: { "Content-Type": "application/json" },
    body:    JSON.stringify(user)
  });
  return await res.json();
}

// ── PUT / PATCH / DELETE ──
await fetch(`/api/users/1`, { method: "PUT",    body: JSON.stringify(data), headers });
await fetch(`/api/users/1`, { method: "PATCH",  body: JSON.stringify(patch), headers });
await fetch(`/api/users/1`, { method: "DELETE" });

// ── Response Methods ──
res.json()     // parse JSON body → object
res.text()     // parse body as plain text
res.blob()     // parse body as binary Blob (files, images)
res.ok         // true if status 200–299
res.status     // e.g. 200, 404, 500
res.headers    // Response Headers object

// ── Abort / Timeout ──
const controller = new AbortController();
setTimeout(() => controller.abort(), 5000);  // cancel after 5s

await fetch("/api/data", { signal: controller.signal });
