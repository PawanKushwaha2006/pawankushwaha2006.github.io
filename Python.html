01. Advanced Data Structures

Master lists, dictionaries, sets, and tuples with comprehensions and advanced operations.

List Comprehensions

# Traditional way
squares = []
for x in range(10):
    squares.append(x**2)

# Pythonic way - List comprehension
squares = [x**2 for x in range(10)]
print(squares)

# With condition
evens = [x for x in range(20) if x % 2 == 0]
print(evens)

# Dictionary comprehension
square_dict = {x: x**2 for x in range(5)}
print(square_dict)

02. File Handling & Exception Management

Learn to read/write files and handle errors gracefully using try-except blocks.

# File Writing
with open('data.txt', 'w') as file:
    file.write("Hello Python!\n")
    file.write("Line 2\n")

# File Reading with Exception Handling
try:
    with open('data.txt', 'r') as file:
        content = file.read()
        print(content)
except FileNotFoundError:
    print("File not found!")
except Exception as e:
    print(f"Error: {e}")
finally:
    print("Operation completed")

# Working with CSV-like data
data = [
    {"name": "Alice", "age": 25},
    {"name": "Bob", "age": 30}
]
for person in data:
    print(f"{person['name']} is {person['age']} years old")

03. Functions: *args, **kwargs & Lambda

Advanced function concepts including variable arguments, keyword arguments, and anonymous functions.

# *args - variable positional arguments
def sum_all(*args):
    return sum(args)

print(sum_all(1, 2, 3, 4, 5)) # 15

# **kwargs - variable keyword arguments
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="Pawan", age=21, city="Delhi")

# Lambda with map, filter
numbers = [1, 2, 3, 4, 5, 6]
squared = list(map(lambda x: x**2, numbers))
print(squared)

evens = list(filter(lambda x: x % 2 == 0, numbers))
print(evens)

04. OOP: Inheritance & Polymorphism

Build complex applications using classes, inheritance, and method overriding.

class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        return "Some sound"
    
    def info(self):
        return f"I am {self.name}"

class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name)
        self.breed = breed
    
    def speak(self): # Method overriding
        return f"{self.name} says Woof!"
    
    def info(self):
        return f"{super().info()}, Breed: {self.breed}"

class Cat(Animal):
    def speak(self):
        return f"{self.name} says Meow!"

# Polymorphism
animals = [Dog("Tommy", "Labrador"), Cat("Whiskers")]
for animal in animals:
    print(animal.speak())
    print(animal.info())

05. Decorators & Generators

Decorators modify function behavior. Generators create iterators efficiently.

# Decorator Example
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"Calling {func.__name__}")
        result = func(*args, **kwargs)
        print(f"Finished {func.__name__}")
        return result
    return wrapper

@logger
def add(a, b):
    return a + b

print(add(5, 3))

# Generator - Memory efficient
def fibonacci(n):
    a, b = 0, 1
    for _ in range(n):
        yield a
        a, b = b, a + b

print("Fibonacci Series:")
for num in fibonacci(10):
    print(num, end=" ")

06. Modules & Virtual Environments

Organize code into modules. Key concepts for real projects.

# Importing modules
import math
import random
from datetime import datetime

# Math operations
print(f"Pi: {math.pi}")

# Random
print(f"Random number: {random.randint(1, 100)}")

# DateTime
now = datetime.now()

# Custom module simulation
class Calculator:
    @staticmethod
    def add(x, y): return x + y
    
    @staticmethod
    def multiply(x, y): return x * y

calc = Calculator()
print(f"5 + 3 = {calc.add(5, 3)}")
print(f"5 * 3 = {calc.multiply(5, 3)}")
