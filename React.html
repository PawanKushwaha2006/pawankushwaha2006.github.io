Lesson 2: useState Hook

useState se component me state rakho. State change hone pe UI auto update hota hai.

function Counter() {
  const [count, setCount] = React.useState(0);

  return (
    <div>
      <h2>Count: {count}</h2>
      <button onClick={() => setCount(count + 1)}>
        Increment
      </button>
    </div>
  );
}

ReactDOM.render(<Counter />, document.getElementById('root'));

Lesson 3: Props

Props se parent se child ko data bhejte hai. Read-only hote hai.

function Greet({ name, age }) {
  return <h2>Hello {name}, age {age}!</h2>
}

function App() {
  return (
    <div>
      <Greet name="Pawan" age={20} />
      <Greet name="Rahul" age={22} />
    </div>
  );
}

ReactDOM.render(<App />, document.getElementById('root'));

Lesson 4: useEffect Hook

useEffect side effects ke liye. Component mount/update pe chalta hai.

function Timer() {
  const [seconds, setSeconds] = React.useState(0);

  React.useEffect(() => {
    const interval = setInterval(() => {
      setSeconds(s => s + 1);
    }, 1000);

    return () => clearInterval(interval);
  }, []);

  return <h2>Timer: {seconds}s</h2>
}

ReactDOM.render(<Timer />, document.getElementById('root'));

Lesson 5: List Rendering

map() se array ko JSX me convert karo. key prop zaruri hai.

function TodoList() {
  const todos = [
    { id: 1, text: 'Learn React' },
    { id: 2, text: 'Build Project' },
    { id: 3, text: 'Deploy App' }
  ];

  return (
    <ul>
      {todos.map(todo => (
        <li key={todo.id}>{todo.text}</li>
      ))}
    </ul>
  );
}

ReactDOM.render(<TodoList />, document.getElementById('root'));
